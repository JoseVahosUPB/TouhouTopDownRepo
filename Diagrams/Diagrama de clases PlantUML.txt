@startuml
skinparam classAttributeIconSize 0
!theme plain
interface IDamageable {
    +TakeDamage(amount: int): void
}

abstract class Entity {
    #health: int
    #speed: float
    +TakeDamage(amount: int): void
    +Move(): void
    +Die(): void
}

class Player {
    -rb: Rigidbody2D
    -currentWeapon: Weapon
    +Move(): void
    +Die(): void
}

abstract class Enemy {
    -damage: int
    -OnTriggerEnter2D(other: Collider2D): void
}

class Chaser {
    +Move(): void
}

class Zigzag {
    -amplitude: float
    -frequency: float
    -initialY: float
    -phase: float
    +Move(): void
}

abstract class Weapon {
    #damage: int
    #fireRate: float
    #bulletPrefab: Bullet
    +Fire(): void
}

class Normal {
    +Fire(): void
}

class Bullet {
    -speed: float
    -damage: int
    -direction: Vector2
    +Initialize(dir: Vector2, damageAmount: int): void
    -OnTriggerEnter2D(other: Collider2D): void
}

' Relaciones de Herencia e Interfaz
Entity ..|> IDamageable
Player --|> Entity
Enemy --|> Entity

Chaser --|> Enemy
Zigzag --|> Enemy

Normal --|> Weapon

' Asociaciones y Dependencias
Player "1" --> "1" Weapon
Weapon ..> Bullet : crea
Bullet ..> IDamageable : daña
@enduml
